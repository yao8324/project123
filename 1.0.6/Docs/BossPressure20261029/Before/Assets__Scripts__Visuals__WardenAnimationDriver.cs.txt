using System;
using System.Collections.Generic;
using ArcaneArchive.Core;
using UnityEngine;
using UnityEngine.Animations;
using UnityEngine.Playables;

namespace ArcaneArchive.Visuals
{
    /// <summary>Battle-clock playback: outgoing poses keep moving throughout the blend.</summary>
    public sealed class WardenAnimationDriver : MonoBehaviour
    {
        public Animator animator;
        public AnimationClip[] clips;
        public Transform castingSocket, leftHand, rightHand;
        public LineRenderer[] rushRibbons;
        public Transform phaseSeal;
        public Transform meleeBladeBase,meleeBladeTip;
        public Material meleeTrailMaterial;
        BladeRibbon meleeRibbon;
        PlayableGraph graph;
        AnimationMixerPlayable mixer;
        AnimationClipPlayable[] players;
        float[] times, weights, origins;
        bool[] loops;
        readonly Dictionary<string,int> indices=new Dictionary<string,int>();
        int current=-1;
        float clock,deathTime,blendAge,blendDuration,moveHold,hitRecoil;
        Transform chest,head;
        public void ReactToHit(){hitRecoil=.12f;}
        Vector3 lastPosition;
        BossEncounter.State lastState;
        static readonly string[] Windups={"VolleyWindup","WaveWindup","FractureRaise","LaserCharge","RushWindup","MeleeWindup"};
        static readonly string[] Releases={"VolleyRelease","WaveRelease","FractureRelease","LaserChannel","RushDrive","MeleeStrike"};
        public string CurrentClip=>current>=0?clips[current].name:"";
        public float SampleTime=>current>=0?times[current]:0;
        public bool IsBlending=>blendAge<blendDuration;
        public float BlendWeightSum { get { float sum=0;if(weights!=null)foreach(float w in weights)sum+=w;return sum;} }

        void Initialize()
        {
            if(graph.IsValid() || !animator || clips==null || clips.Length==0)return;
            animator.applyRootMotion=false;animator.cullingMode=AnimatorCullingMode.AlwaysAnimate;
            graph=PlayableGraph.Create("Warden continuous battle animation");graph.SetTimeUpdateMode(DirectorUpdateMode.Manual);
            mixer=AnimationMixerPlayable.Create(graph,clips.Length);
            players=new AnimationClipPlayable[clips.Length];times=new float[clips.Length];weights=new float[clips.Length];origins=new float[clips.Length];loops=new bool[clips.Length];
            indices.Clear();
            for(int i=0;i<clips.Length;i++)
            {
                if(!clips[i])continue;
                players[i]=AnimationClipPlayable.Create(graph,clips[i]);players[i].SetSpeed(0);
                players[i].SetApplyFootIK(false);players[i].SetApplyPlayableIK(false);graph.Connect(players[i],0,mixer,i);
                string key=clips[i].name.Substring(clips[i].name.LastIndexOf("AA_",StringComparison.Ordinal)+3);
                indices[key]=i;
            }
            AnimationPlayableOutput.Create(graph,"Skeletal output",animator).SetSourcePlayable(mixer);
            graph.Play();lastPosition=transform.position;
            if(meleeTrailMaterial && meleeBladeBase && meleeBladeTip)
            {var go=new GameObject("Warden blade arc");go.transform.SetParent(transform,false);meleeRibbon=go.AddComponent<BladeRibbon>();meleeRibbon.Initialize(meleeTrailMaterial);}
            foreach(var bone in animator.GetComponentsInChildren<Transform>()){if(bone.name=="chest")chest=bone;if(bone.name=="head")head=bone;}
        }
        public void ResetPose()
        {
            Initialize();current=-1;clock=deathTime=blendAge=moveHold=hitRecoil=0;lastState=BossEncounter.State.Inactive;lastPosition=transform.position;
            if(!graph.IsValid())return;
            if(meleeRibbon)meleeRibbon.ResetTrail();
            Array.Clear(weights,0,weights.Length);Array.Clear(times,0,times.Length);
            Present(0,1,0,BossEncounter.State.Intro,0,1);
        }
        public void Present(float dt,int phase,int attack,BossEncounter.State state,float elapsed,float warning,float recovery=1.15f)
        {
            Initialize();if(!graph.IsValid())return;
            clock+=dt;
            float speed=dt>0?(transform.position-lastPosition).magnitude/dt:0;
            lastPosition=transform.position;
            moveHold=speed>.08f?.16f:Mathf.Max(0,moveHold-dt);
            if(state==BossEncounter.State.Won){if(lastState!=state)deathTime=0;deathTime+=dt;}
            string name="Idle";float t=clock;bool loop=true;
            if(state==BossEncounter.State.Intro){name="Intro";t=elapsed/2.2f;loop=false;}
            else if(state==BossEncounter.State.Observe)name=moveHold>0?"Glide":"Idle";
            else if(state==BossEncounter.State.Telegraph){name=Windups[Mathf.Clamp(attack,0,5)];t=Mathf.Clamp01(elapsed/Mathf.Max(.01f,warning));loop=false;}
            else if(state==BossEncounter.State.Attack)
            {
                name=Releases[Mathf.Clamp(attack,0,5)];t=elapsed;loop=false;
                if(attack==5)t=elapsed/.72f;
                if(attack==1 && phase==2 && elapsed>=1.25f)t-=1.25f;
            }
            else if(state==BossEncounter.State.Recovery){name="Recovery";t=Mathf.Clamp01(elapsed/Mathf.Max(.1f,recovery));loop=false;}
            else if(state==BossEncounter.State.Stagger){name="Stagger";t=elapsed;loop=false;}
            else if(state==BossEncounter.State.PhaseBreak){name="PhaseBreak";t=elapsed/2.2f;loop=false;}
            else if(state==BossEncounter.State.Won){name="Defeat";t=deathTime;loop=false;}
            int next;if(!indices.TryGetValue(name,out next))return;
            if(next!=current)
            {
                // Preserve the evaluated mixture even when a second transition interrupts the first.
                Array.Copy(weights,origins,weights.Length);bool first=current<0;current=next;blendAge=0;
                blendDuration=first?0:state==BossEncounter.State.Recovery?Mathf.Min(recovery*.45f,.42f):state==BossEncounter.State.Attack?.10f:state==BossEncounter.State.Telegraph?.18f:.24f;
            }
            for(int i=0;i<clips.Length;i++)
            {
                if(i==current || weights[i]<=0 || !clips[i])continue;
                times[i]+=dt;times[i]=loops[i]?Mathf.Repeat(times[i],clips[i].length):Mathf.Min(times[i],clips[i].length);
                players[i].SetTime(times[i]);
            }
            float length=clips[current].length;
            if(state==BossEncounter.State.Telegraph || state==BossEncounter.State.Recovery || state==BossEncounter.State.Intro || state==BossEncounter.State.PhaseBreak || (state==BossEncounter.State.Attack && attack==5))t*=length;
            times[current]=loop?Mathf.Repeat(t,Mathf.Max(.001f,length)):Mathf.Clamp(t,0,length);loops[current]=loop;
            players[current].SetTime(times[current]);blendAge+=dt;
            float blend=blendDuration<=0?1:Mathf.SmoothStep(0,1,Mathf.Clamp01(blendAge/blendDuration));
            for(int i=0;i<clips.Length;i++)
            {weights[i]=Mathf.Lerp(origins[i],i==current?1:0,blend);mixer.SetInputWeight(i,weights[i]);}
            graph.Evaluate(0);lastState=state;
            // The graph restores the skeletal pose before this small additive response.
            // It never moves the gameplay root or changes the weapon grip.
            if(hitRecoil>0)
            {
                float impulse=Mathf.Sin(Mathf.Clamp01(hitRecoil/.12f)*Mathf.PI);
                if(chest)chest.localRotation*=Quaternion.Euler(-4*impulse,0,2*impulse);
                if(head)head.localRotation*=Quaternion.Euler(2*impulse,0,-impulse);
                hitRecoil=Mathf.Max(0,hitRecoil-dt);
            }
            if(meleeRibbon)meleeRibbon.Tick(dt,state==BossEncounter.State.Attack && attack==5 && elapsed>.10f && elapsed<.46f,meleeBladeBase.position,meleeBladeTip.position);
            if(chest && state!=BossEncounter.State.Won && state!=BossEncounter.State.Stagger)
                chest.localRotation*=Quaternion.Euler(Mathf.Sin(clock*2.1f)*.38f,0,Mathf.Sin(clock*1.3f)*.24f);
            if(phaseSeal)
            {
                phaseSeal.gameObject.SetActive(state==BossEncounter.State.PhaseBreak);
                phaseSeal.localRotation=Quaternion.Euler(90,clock*13,0);
                phaseSeal.localScale=Vector3.one*(1.3f+.8f*Mathf.Sin(Mathf.Clamp01(elapsed/2.2f)*Mathf.PI));
            }
            // Serialized legacy lines remain disabled; the encounter owns the world-space wake.
            if(rushRibbons!=null)foreach(var line in rushRibbons)if(line)line.enabled=false;
        }
        void OnDestroy(){if(graph.IsValid())graph.Destroy();}
    }
}
