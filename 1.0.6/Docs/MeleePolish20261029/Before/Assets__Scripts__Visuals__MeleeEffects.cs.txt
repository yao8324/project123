using ArcaneArchive.Core;
using UnityEngine;
using UnityEngine.Rendering;

namespace ArcaneArchive.Visuals
{
    // Short, clock-driven sparks. No independent particle simulation during pause.
    public sealed class MeleeEffects : MonoBehaviour
    {
        BossEncounter encounter;
        LineRenderer[] rays;
        Vector3[] directions;
        float age, lifetime;
        bool ring, sweep;
        Vector3 sweepForward;
        float sweepRadius, sweepArc;
        public static LineRenderer CreateLine(string name,Transform parent,Material material,float width,int count)
        {
            var go=new GameObject(name);go.transform.SetParent(parent,false);go.layer=parent.gameObject.layer;
            var l=go.AddComponent<LineRenderer>();l.sharedMaterial=material;l.useWorldSpace=true;l.positionCount=count;
            l.startWidth=l.endWidth=width;l.numCapVertices=3;l.numCornerVertices=2;
            l.shadowCastingMode=ShadowCastingMode.Off;l.receiveShadows=false;return l;
        }
        public static void Burst(Transform parent,Vector3 point,Vector3 normal,Material material,BossEncounter owner,bool parry)
        {
            var go=new GameObject(parry?"Shield parry seal":"Steel and seal impact");go.transform.SetParent(parent,false);go.transform.position=point;go.layer=parent.gameObject.layer;
            var fx=go.AddComponent<MeleeEffects>();fx.encounter=owner;fx.lifetime=parry?.44f:.26f;fx.ring=parry;
            fx.rays=new LineRenderer[parry?13:10];fx.directions=new Vector3[fx.rays.Length];
            Quaternion basis=Quaternion.LookRotation(normal.sqrMagnitude>.001f?normal:Vector3.forward);
            for(int i=0;i<fx.rays.Length;i++)
            {
                float a=i*2.399963f;
                fx.directions[i]=basis*new Vector3(Mathf.Cos(a),Mathf.Sin(a),-.35f).normalized;
                fx.rays[i]=CreateLine("Gilt spark",go.transform,material,parry?.022f:.014f,parry&&i==12?49:2);
                for(int j=0;j<fx.rays[i].positionCount;j++)fx.rays[i].SetPosition(j,point);
            }
        }
        public static void Sweep(Transform parent,Vector3 center,Vector3 forward,float radius,float arc,Material material,BossEncounter owner)
        {
            var go=new GameObject("Melee seal arc");go.transform.SetParent(parent,false);go.transform.position=center;
            var fx=go.AddComponent<MeleeEffects>();fx.encounter=owner;fx.lifetime=.32f;fx.sweep=true;fx.sweepRadius=radius;fx.sweepArc=arc;fx.sweepForward=forward;
            fx.rays=new[]{CreateLine("Seal cutting edge",go.transform,material,.055f,49)};
            for(int i=0;i<49;i++)fx.rays[0].SetPosition(i,center);
        }
        void Update()
        {
            if(encounter && encounter.IsPaused)return;
            if(encounter && encounter.IsFinished){Destroy(gameObject);return;}
            age+=Mathf.Min(Time.deltaTime,.1f);float t=age/lifetime;
            if(t>=1){Destroy(gameObject);return;}
            if(sweep)
            {
                var line=rays[0];line.startColor=line.endColor=new Color(1,1,1,1-t);
                for(int i=0;i<49;i++)
                {
                    float a=Mathf.Lerp(-sweepArc*.5f,sweepArc*.5f,i/48f);
                    line.SetPosition(i,transform.position+Quaternion.AngleAxis(a,Vector3.up)*sweepForward*sweepRadius*Mathf.Lerp(.82f,1,t));
                }
                return;
            }
            for(int i=0;i<rays.Length;i++)
            {
                var l=rays[i];l.startColor=l.endColor=new Color(1,1,1,1-t);
                if(ring && i==12)
                {
                    Vector3 side=Vector3.Cross(directions[0],directions[1]).normalized;
                    Vector3 up=Vector3.Cross(side,directions[0]).normalized;
                    for(int j=0;j<49;j++){float a=j*Mathf.PI*2/48;l.SetPosition(j,transform.position+(directions[0]*Mathf.Cos(a)+up*Mathf.Sin(a))*(.1f+t*.5f));}
                }
                else
                {
                    float distance=t*(.35f+(i%3)*.14f);
                    l.SetPosition(0,transform.position+directions[i]*distance);
                    l.SetPosition(1,transform.position+directions[i]*(distance+.09f*(1-t)));
                }
            }
        }
    }
}
