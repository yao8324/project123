using ArcaneArchive.Core;
using ArcaneArchive.UI;
using ArcaneArchive.Visuals;
using UnityEngine;

namespace ArcaneArchive.Player
{
    // Rigid weapons and their gripping hands move together; gameplay owns all timing.
    public sealed class SwordShieldView : MonoBehaviour
    {
        public Transform swordPivot, shieldPivot, bladeBase, bladeTip;
        public PlayerMelee melee;
        public Camera presentationCamera, viewCamera;
        FirstPersonController movement;
        PlayerCombat combat;
        RecoveryMenu menu;
        BossEncounter encounter;
        Renderer[] renderers;
        BladeRibbon ribbon;
        float bobClock,guardWeight,recoil;
        Vector3 swordPosition, shieldPosition;
        Quaternion swordRotation, shieldRotation;
        static readonly Vector3 SwordRest=new Vector3(.30f,-.31f,.57f), ShieldRest=new Vector3(-.35f,-.32f,.57f);
        void Start()
        {
            if(!melee)melee=GetComponentInParent<PlayerMelee>();
            movement=melee.GetComponent<FirstPersonController>();combat=melee.GetComponent<PlayerCombat>();menu=melee.GetComponent<RecoveryMenu>();encounter=melee.GetComponent<BossEncounter>();
            var go=new GameObject("Blade light ribbon");go.transform.SetParent(transform,false);go.layer=gameObject.layer;
            ribbon=go.AddComponent<BladeRibbon>();ribbon.Initialize(melee.loadout.goldEffect);
            renderers=GetComponentsInChildren<Renderer>(true);ResetPose();
        }
        public void ResetPose()
        {
            swordPosition=SwordRest;shieldPosition=ShieldRest;swordRotation=Quaternion.Euler(24,8,-18);shieldRotation=Quaternion.Euler(5,24,-13);
            guardWeight=recoil=0;if(ribbon)ribbon.ResetTrail();ApplyPose();
        }
        public void BeginSwing(){if(ribbon)ribbon.ResetTrail();}
        public void Interrupt(){recoil=.45f;if(ribbon)ribbon.ResetTrail();}
        public void Impact(bool parry)
        {
            recoil=parry?1:.55f;
            if(parry && melee && melee.loadout)MeleeEffects.Burst(transform,shieldPivot.position+viewCamera.transform.forward*.12f,-viewCamera.transform.forward,melee.loadout.goldEffect,encounter,true);
        }
        void ApplyPose()
        {
            if(swordPivot)swordPivot.SetLocalPositionAndRotation(swordPosition,swordRotation);
            if(shieldPivot)shieldPivot.SetLocalPositionAndRotation(shieldPosition,shieldRotation);
        }
        void LateUpdate()
        {
            if(!movement || !melee.loadout)return;
            bool visible=!menu.IsOpen && (!encounter || !encounter.IsFinished);
            foreach(var r in renderers)if(r)r.enabled=visible;
            if(!visible || movement.SimulationPaused)return;
            float dt=Mathf.Min(Time.deltaTime,.1f);bobClock+=dt*(movement.IsMoving?8:1.9f);recoil=Mathf.Max(0,recoil-dt*4);
            if(presentationCamera && viewCamera){presentationCamera.fieldOfView=viewCamera.fieldOfView;presentationCamera.aspect=viewCamera.aspect;}
            float guard=(combat.Guarding || melee.PracticeGuarding)?1:0;
            guardWeight=Mathf.Lerp(guardWeight,guard,1-Mathf.Exp(-18*dt));
            float movementWeight=Mathf.Clamp01(movement.PlanarSpeed/4);
            Vector3 sway=new Vector3(Mathf.Sin(bobClock)*.010f,Mathf.Cos(bobClock*2)*.007f,0)*movementWeight;
            sway.y+=Mathf.Sin(bobClock*.7f)*.0028f;
            Vector3 rp=SwordRest, rr=new Vector3(24,8,-18),lp=ShieldRest,lr=new Vector3(5,24,-13);
            if(melee.Busy)SwingPose(melee.Action,melee.Combo,melee.Progress,ref rp,ref rr,ref lp,ref lr);
            else if(movement.IsSprinting){rp+=new Vector3(.02f,-.10f,-.03f);rr+=new Vector3(35,0,24);lp.y-=.06f;}
            lp=Vector3.Lerp(lp,new Vector3(-.11f,-.13f,.66f),guardWeight);lr=Vector3.Lerp(lr,new Vector3(-4,7,-4),guardWeight);
            lp.z-=recoil*.045f;rp.z-=recoil*.018f;
            float blend=1-Mathf.Exp(-28*dt);
            swordPosition=Vector3.Lerp(swordPosition,rp+sway,blend);shieldPosition=Vector3.Lerp(shieldPosition,lp+sway*.75f,blend);
            swordRotation=Quaternion.Slerp(swordRotation,Quaternion.Euler(rr),blend);shieldRotation=Quaternion.Slerp(shieldRotation,Quaternion.Euler(lr),blend);ApplyPose();
            bool trail=melee.Busy && melee.Action!=2 && melee.Progress>.27f && melee.Progress<.75f;
            if(bladeBase && bladeTip)ribbon.Tick(dt,trail,bladeBase.position,bladeTip.position);
        }
        static void SwingPose(int action,int combo,float t,ref Vector3 p,ref Vector3 r,ref Vector3 l,ref Vector3 lr)
        {
            Vector3 restP=p,restR=r,windP,windR,cutP,cutR,followP,followR;
            if(action==2)
            {
                float drive=Mathf.Sin(Mathf.Clamp01(t/.78f)*Mathf.PI);
                l+=new Vector3(.15f,.13f,.25f)*drive;lr=Vector3.Lerp(lr,new Vector3(0,0,-5),drive);
                p+=new Vector3(.05f,-.04f,-.08f)*drive;r+=new Vector3(10,5,15)*drive;return;
            }
            if(action==3)
            {
                windP=new Vector3(.25f,-.29f,.42f);windR=new Vector3(88,4,-6);
                cutP=new Vector3(.05f,-.13f,1.04f);cutR=new Vector3(88,-4,0);
                followP=new Vector3(.12f,-.20f,.84f);followR=new Vector3(72,0,-10);
            }
            else if(action==1 || (action==0 && combo==2))
            {
                windP=new Vector3(.26f,.05f,.46f);windR=new Vector3(-50,-12,-6);
                cutP=new Vector3(.03f,-.27f,.83f);cutR=new Vector3(112,0,6);
                followP=new Vector3(-.04f,-.42f,.64f);followR=new Vector3(140,-12,25);
            }
            else if(action==0 && combo==1)
            {
                windP=new Vector3(-.19f,-.28f,.49f);windR=new Vector3(44,-30,68);
                cutP=new Vector3(.25f,-.10f,.82f);cutR=new Vector3(65,25,-65);
                followP=new Vector3(.43f,-.22f,.61f);followR=new Vector3(40,45,-90);
            }
            else
            {
                windP=new Vector3(.45f,-.13f,.44f);windR=new Vector3(-12,30,-72);
                cutP=new Vector3(-.16f,-.16f,.83f);cutR=new Vector3(76,-30,64);
                followP=new Vector3(-.29f,-.34f,.56f);followR=new Vector3(105,-55,96);
                if(action==4){windR.y=70;cutP.x=-.28f;followP.x=-.42f;followR.y=-95;}
            }
            float phase=t<.28f?t/.28f:t<.53f?(t-.28f)/.25f:t<.73f?(t-.53f)/.20f:(t-.73f)/.27f;
            phase=phase*phase*(3-2*phase);
            if(t<.28f){p=Vector3.Lerp(restP,windP,phase);r=Vector3.Lerp(restR,windR,phase);}
            else if(t<.53f){p=Vector3.Lerp(windP,cutP,phase);r=Vector3.Lerp(windR,cutR,phase);}
            else if(t<.73f){p=Vector3.Lerp(cutP,followP,phase);r=Vector3.Lerp(cutR,followR,phase);}
            else {p=Vector3.Lerp(followP,restP,phase);r=Vector3.Lerp(followR,restR,phase);}
            float support=Mathf.Sin(t*Mathf.PI);l+=new Vector3(.035f,.025f,-.07f)*support;lr.z-=support*7;
        }
    }
}
