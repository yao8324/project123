    static void SurveyPositions()
    {
        Physics.SyncTransforms();var controller=Object.FindFirstObjectByType<FirstPersonController>().GetComponent<CharacterController>();var rows=new List<PositionProbe>();
        foreach(var point in new[]{new Vector3(5.8f,.08f,7.3f),new Vector3(5.2f,.08f,7.1f),new Vector3(5.05f,.08f,7),new Vector3(7.325f,.08f,6.9f),new Vector3(7.325f,.08f,10),new Vector3(4.275f,.08f,10),new Vector3(4.275f,.08f,6.9f),new Vector3(6,.08f,1.4f)})
        {
            bool supported=Physics.Raycast(point+Vector3.up,Vector3.down,out var hit,2,Physics.DefaultRaycastLayers,QueryTriggerInteraction.Ignore);
            var feet=supported?new Vector3(point.x,hit.point.y+.08f,point.z):point;
            float half=controller.height*.5f-controller.radius;var center=feet+controller.center;
            var overlaps=Physics.OverlapCapsule(center-Vector3.up*half,center+Vector3.up*half,controller.radius+.02f,Physics.DefaultRaycastLayers,QueryTriggerInteraction.Ignore).Where(c=>c!=controller&&!c.transform.IsChildOf(controller.transform)).Select(c=>c.name).Distinct().ToArray();
            rows.Add(new PositionProbe{requested=point,supported=supported,floorY=supported?hit.point.y:-999,floor=supported?hit.collider.name:"NONE",occupied=overlaps});
        }
        File.WriteAllText(Out+"/position-survey.json",JsonUtility.ToJson(new PositionSurvey{probes=rows.ToArray()},true));
    }

    static IEnumerator<Gate> Sequence()

        foreach(var anchor in workstation.GetComponentsInChildren<MaintenanceArtAnchor>().Where(a=>a.anchorId.StartsWith("ARCHIVE_RECORD_SLOT_")))
        {
            foreach(float x in new[]{-.14f,.14f})foreach(float z in new[]{-.18f,.18f})
            {
                var from=anchor.transform.TransformPoint(new Vector3(x,.006f,z));
                Check(Physics.Raycast(from,Vector3.down,out var hit,.08f,Physics.DefaultRaycastLayers,QueryTriggerInteraction.Ignore) && Mathf.Abs(hit.point.y-.836f)<.025f,"Actual production desk supports record corner "+anchor.anchorId);
            }
            foreach(var gate in Interact(anchor,"Read "+anchor.anchorId))yield return gate;
        }
        foreach(var gate in Interact(WorkAnchor("ARCHIVE_WORK_BADGE"),"Pre-attack work badge"))yield return gate;
        Capture("02_production_badge");
        foreach(var gate in Interact(WorkAnchor("ARCHIVE_TIE_CORD"),"Tie the archival seal"))yield return gate;
        Check(workstation.IsArchiveSealed,"Cord presentation changes only after its interaction callback");
        Capture("03_production_sealed");
        scenario="Repeat sealed cord press";int requestCount=requests.Count;Keys(Key.E);yield return Wait(.1f);Keys();yield return Wait(.2f);
        Check(requests.Count==requestCount,"Sealed cord no longer receives repeated loose-cord E input");
        foreach(var p in new[]{new Vector3(5.2f,0,6.9f),new Vector3(7.325f,0,6.9f),new Vector3(7.325f,0,10.0f),new Vector3(4.275f,0,10.0f),new Vector3(4.275f,0,6.9f),new Vector3(5.2f,0,7.1f)})

    static void Tick()
    {
        if(!EditorApplication.isPlaying || finished || frame==Time.frameCount)return;frame=Time.frameCount;
        try
        {
            if(Time.realtimeSinceStartup-startedAt>600)throw new TimeoutException("Overall route exceeded ten minutes");
            if(menu.IsOpen)menu.Resume();player.SetCursorLocked(true);player.SetGameplayInputEnabled(true);
            minY=Mathf.Min(minY,player.transform.position.y);if(player.transform.position.y<-.15f)throw new Exception("Player fell below the library/maintenance floor");
            var action=(InputAction)typeof(FirstPersonController).GetField("moveAction",Private).GetValue(player);
            var p=player.transform.position;trace.WriteLine(string.Join(",",Time.frameCount,(Time.realtimeSinceStartup-startedAt).ToString("F4"),scenario,p.x.ToString("F4"),p.y.ToString("F4"),p.z.ToString("F4"),player.IsGrounded,player.IsCrouching,player.IsSprinting,player.GameplayInputEnabled,typeof(FirstPersonController).GetField("waitForInputRelease",Private).GetValue(player),typeof(FirstPersonController).GetField("inputReleaseFrames",Private).GetValue(player),action.ReadValue<Vector2>().ToString().Replace(',',';'),string.Join("+",held),Keyboard.current?.name,action.enabled));
            if(Time.realtimeSinceStartup-lastCapture>=.125f){Capture(null,true);times.WriteLine(captureIndex+","+(Time.realtimeSinceStartup-startedAt).ToString("F4")+","+scenario);captureIndex++;lastCapture=Time.realtimeSinceStartup;}
            if(gate!=null)
            {
                gate.tick?.Invoke();if(Time.realtimeSinceStartup>gate.deadline)throw new TimeoutException("Blocked in "+scenario+" at "+player.transform.position);
                if(!gate.ready())return;gate=null;
            }
            if(sequence.MoveNext())gate=sequence.Current;
        }catch(Exception ex){Fail(ex);}
    }
    static void Capture(string name,bool video=false)
    {
        var previous=view.targetTexture;var active=RenderTexture.active;
        try{view.targetTexture=target;view.Render();RenderTexture.active=target;texture.ReadPixels(new Rect(0,0,1280,720),0,0);texture.Apply();File.WriteAllBytes(video?Out+"/Frames/"+captureIndex.ToString("D5")+".jpg":Out+"/Captures/"+name+".png",video?texture.EncodeToJPG(80):texture.EncodeToPNG());}
        finally{view.targetTexture=previous;RenderTexture.active=active;}
    }
    static void Write()
    {
        Directory.CreateDirectory(Out);File.WriteAllText(Out+"/runtime-validation.json",JsonUtility.ToJson(new Report{complete=finished&&string.IsNullOrEmpty(failure),scenario=scenario,failure=failure,unity=Application.unityVersion,device=SystemInfo.graphicsDeviceName,cpu=SystemInfo.processorType,pipeline=UnityEngine.Rendering.GraphicsSettings.currentRenderPipeline?.name,quality=QualitySettings.names[QualitySettings.GetQualityLevel()],checksPassed=checks.Count,videoFrames=captureIndex,roundTrips=roundTrip,elapsedSeconds=Time.realtimeSinceStartup-startedAt,minFeetY=minY,checks=checks.ToArray(),interactionRequests=requests.ToArray()},true));trace?.Flush();times?.Flush();
    }
    static void Fail(Exception ex)
    {
        failure=ex.ToString();finished=true;Debug.LogError(failure);try{if(view && texture)Capture("FAILURE");Write();}finally{Stop();}
    }
    static void Stop()
    {
        Keys();InputSystem.onBeforeUpdate-=Input;EditorApplication.update-=Tick;scope?.Dispose();trace?.Dispose();times?.Dispose();SessionState.SetBool(Flag,false);
        if(keyboard!=null && keyboard.added)InputSystem.RemoveDevice(keyboard);if(mouse!=null && mouse.added)InputSystem.RemoveDevice(mouse);
        Debug.Log("[ArchiveStage2] FINISHED "+scenario);EditorApplication.Exit(string.IsNullOrEmpty(failure)?0:2);
    }
