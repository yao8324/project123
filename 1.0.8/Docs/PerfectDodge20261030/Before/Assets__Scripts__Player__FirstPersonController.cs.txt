using UnityEngine;
using UnityEngine.InputSystem;

namespace ArcaneArchive.Player
{
    [RequireComponent(typeof(CharacterController))]
    public sealed class FirstPersonController : MonoBehaviour
    {
        [Header("Input")]
        [SerializeField] private InputActionAsset inputActions;
        [SerializeField] private Transform cameraPivot;

        [Header("Movement")]
        [SerializeField, Min(0f)] private float walkSpeed = 4f;
        [SerializeField, Min(1f)] private float sprintMultiplier = 1.5f;
        [SerializeField, Min(0f)] private float jumpSpeed = 7f;
        [SerializeField] private float gravity = -20f;

        [Header("Crouch")]
        [SerializeField, Min(0.1f)] private float crouchHeight = 1f;
        [SerializeField, Min(0f)] private float crouchTransitionSpeed = 8f;
        [SerializeField] private LayerMask ceilingLayers = Physics.DefaultRaycastLayers;

        [Header("Ground Check")]
        [SerializeField, Min(0f)] private float groundCheckOriginHeight = 0.15f;
        [SerializeField, Min(0f)] private float groundCheckDistance = 0.35f;
        [SerializeField] private LayerMask groundLayers = Physics.DefaultRaycastLayers;

        [Header("Look")]
        [SerializeField, Min(0f)] private float mouseSensitivity = 0.08f;
        [SerializeField, Range(1f, 89f)] private float maxPitch = 80f;

        public float MouseSensitivityMultiplier => mouseSensitivity / .08f;
        public void SetMouseSensitivityMultiplier(float multiplier)
        {
            if (float.IsNaN(multiplier) || float.IsInfinity(multiplier)) return;
            mouseSensitivity = .08f * Mathf.Clamp(multiplier, .25f, 3f);
        }


        private CharacterController characterController;
        private InputActionMap playerActions;
        private InputActionAsset runtimeInputActions;
        private ArcaneArchive.UI.RecoveryMenu pauseMenu;
        private ArcaneArchive.Core.BossEncounter encounter;
        public bool SimulationPaused => (pauseMenu && pauseMenu.IsOpen) || (encounter && encounter.IsPaused);
        public ArcaneArchive.Core.LibraryInputBindings Bindings { get; private set; }
        private InputAction moveAction;
        private InputAction lookAction;
        private InputAction sprintAction;
        private InputAction jumpAction;
        private InputAction crouchAction;
        private float verticalVelocity;
        private float pitch;
        private float standingHeight;
        private Vector3 standingCenter;
        private Vector3 standingCameraLocalPosition;
        private bool isOnGroundSurface;
        private bool wasOnGroundSurface;
        private bool jumpConsumed;
        private bool waitForInputRelease;
        private int inputReleaseFrames;
        private bool MovementInputAllowed => GameplayInputEnabled && !waitForInputRelease;


        // Read-only presentation state, measured from CharacterController displacement.
        private bool groundSampleInitialized;
        private CollisionFlags frameCollisionFlags;
        public Vector3 ActualVelocity { get; private set; }
        public float PlanarSpeed { get; private set; }
        public bool IsMoving => PlanarSpeed > 0.05f;
        public bool IsSprinting { get; private set; }
        public bool IsCrouching => characterController != null && characterController.height < standingHeight - 0.01f;
        public bool IsGrounded { get; private set; }
        public uint JumpSequence { get; private set; }
        public uint LandSequence { get; private set; }

        public bool GameplayInputEnabled { get; private set; } = true;
#if UNITY_EDITOR
        private bool batchCursorLocked;
#endif
        public bool CursorIsLocked
        {
            get
            {
#if UNITY_EDITOR
                if(Application.isBatchMode)return batchCursorLocked;
#endif
                return Cursor.lockState==CursorLockMode.Locked;
            }
        }


        private void Awake()
        {
            characterController = GetComponent<CharacterController>();
            pauseMenu = GetComponent<ArcaneArchive.UI.RecoveryMenu>();
            encounter = GetComponent<ArcaneArchive.Core.BossEncounter>();
            standingHeight = characterController.height;
            standingCenter = characterController.center;

            if (cameraPivot != null)
            {
                standingCameraLocalPosition = cameraPivot.localPosition;
                pitch = Mathf.Clamp(Mathf.DeltaAngle(0f, cameraPivot.localEulerAngles.x), -maxPitch, maxPitch);
            }

            ConfigureActions();
        }

        private void OnEnable()
        {
            playerActions?.Enable();
        }

        private void Start()
        {
            SetCursorLocked(true);
        }

        private void OnDisable()
        {
            playerActions?.Disable();
            SetCursorLocked(false);
        }

        private void Update()
        {
            // Preserve position, gravity velocity, crouch and jump/landing state during a pause.
            // Input suppression alone must not allow gravity to advance behind the menu.
            if (SimulationPaused)
            {
                ActualVelocity = Vector3.zero; PlanarSpeed = 0f; IsSprinting = false;
                return;
            }
            if (inputReleaseFrames > 0) inputReleaseFrames--;
            else if (waitForInputRelease && moveAction.ReadValue<Vector2>().sqrMagnitude < .001f &&
                !sprintAction.IsPressed() && !jumpAction.IsPressed() && !crouchAction.IsPressed() &&
                Mouse.current?.leftButton.isPressed != true) waitForInputRelease = false;
            Vector3 previousPosition = transform.position;
            frameCollisionFlags = CollisionFlags.None;
            UpdateGroundState();
            HandleCursorLock();
            ApplyGravity();

            if (MovementInputAllowed)
            {
                UpdateCrouch();
                UpdateLook();
                UpdateMovement();
            }

            ActualVelocity = Time.deltaTime > 0f ? (transform.position - previousPosition) / Time.deltaTime : Vector3.zero;
            PlanarSpeed = Vector3.ProjectOnPlane(ActualVelocity, Vector3.up).magnitude;
            IsSprinting = MovementInputAllowed && IsMoving && sprintAction != null && sprintAction.IsPressed();
            // Move is called separately for gravity and horizontal movement. The last
            // call alone can lose the Below contact observed by the first call.
            bool groundedNow = (frameCollisionFlags & CollisionFlags.Below) != 0 && verticalVelocity <= 0f;
            if (groundSampleInitialized && groundedNow && !IsGrounded) LandSequence++;
            IsGrounded = groundedNow;
            groundSampleInitialized = true;
        }

        public void SetGameplayInputEnabled(bool enabled)
        {
            GameplayInputEnabled = enabled;
        }

        public void SetCursorLocked(bool locked)
        {
#if UNITY_EDITOR
            // Batch acceptance uses a virtual lock; never capture the user's desktop mouse.
            if(Application.isBatchMode){batchCursorLocked=locked;return;}
#endif
            Cursor.lockState = locked ? CursorLockMode.Locked : CursorLockMode.None;
            Cursor.visible = !locked;
        }

        public void WaitForReleasedControls() { waitForInputRelease = true; inputReleaseFrames = 2; }

        public bool TryReturnToSpawn(Vector3 spawn, Quaternion facing)
        {
            if (characterController == null || !characterController.enabled) return false;
            Vector3[] offsets = { Vector3.zero, Vector3.right*.6f, Vector3.left*.6f,
                Vector3.forward*.6f, Vector3.back*.6f };
            foreach (Vector3 offset in offsets)
            {
                Vector3 probe = spawn + offset;
                if (!Physics.Raycast(probe + Vector3.up, Vector3.down, out RaycastHit floor, 2f,
                    groundLayers, QueryTriggerInteraction.Ignore) || floor.normal.y < .75f) continue;
                Vector3 feet = new Vector3(probe.x, floor.point.y + .08f, probe.z);
                Vector3 center = feet + facing * standingCenter;
                float half = standingHeight*.5f-characterController.radius;
                var occupied = Physics.OverlapCapsule(center-Vector3.up*half,center+Vector3.up*half,
                    characterController.radius+.02f,ceilingLayers,QueryTriggerInteraction.Ignore);
                bool clear = true;
                foreach(var hit in occupied)
                    if(hit!=characterController && !hit.transform.IsChildOf(transform)){clear=false;break;}
                if(!clear)continue;
                characterController.enabled=false;
                try
                {
                    transform.SetPositionAndRotation(feet,facing);
                    characterController.height=standingHeight;characterController.center=standingCenter;
                    pitch=0f;
                    if(cameraPivot){cameraPivot.localPosition=standingCameraLocalPosition;cameraPivot.localRotation=Quaternion.identity;}
                    verticalVelocity=0f;jumpConsumed=false;isOnGroundSurface=false;wasOnGroundSurface=false;
                    groundSampleInitialized=false;IsGrounded=false;frameCollisionFlags=CollisionFlags.None;
                    ActualVelocity=Vector3.zero;PlanarSpeed=0f;IsSprinting=false;waitForInputRelease=true;inputReleaseFrames=2;
                }
                finally{characterController.enabled=true;}
                foreach(var view in GetComponentsInChildren<FirstPersonViewModel>(true))view.ResetPresentation();
                return true;
            }
            return false;
        }

        private void ConfigureActions()
        {
            if (inputActions == null)
            {
                Debug.LogError("FirstPersonController requires the project InputActionAsset.", this);
                return;
            }

            runtimeInputActions = Instantiate(inputActions);
            Bindings = GetComponent<ArcaneArchive.Core.LibraryInputBindings>();
            if(!Bindings) Bindings = gameObject.AddComponent<ArcaneArchive.Core.LibraryInputBindings>();
            Bindings.Initialize(runtimeInputActions);
            playerActions = runtimeInputActions.FindActionMap("Player", throwIfNotFound: true);
            moveAction = playerActions.FindAction("Move", throwIfNotFound: true);
            lookAction = playerActions.FindAction("Look", throwIfNotFound: true);
            sprintAction = playerActions.FindAction("Sprint", throwIfNotFound: true);
            jumpAction = playerActions.FindAction("Jump", throwIfNotFound: true);
            crouchAction = playerActions.FindAction("Crouch", throwIfNotFound: true);
        }

        private void HandleCursorLock()
        {
            if(Bindings && !Bindings.SuppressInputThisFrame && GameplayInputEnabled && !CursorIsLocked &&
                Bindings.RelockAction.WasPressedThisFrame()) SetCursorLocked(true);
        }

        private void OnDestroy()
        {
            if(Bindings)Bindings.Cancel();
            if(runtimeInputActions)Destroy(runtimeInputActions);
        }

        private void ApplyGravity()
        {
            if (isOnGroundSurface && verticalVelocity < 0f)
            {
                verticalVelocity = -2f;
            }

            if (MovementInputAllowed && isOnGroundSurface && !jumpConsumed && jumpAction.WasPressedThisFrame())
            {
                verticalVelocity = jumpSpeed;
                jumpConsumed = true;
                JumpSequence++;
            }

            // Integrate the ballistic displacement before changing velocity. A long
            // render frame must not turn a normal jump into a tiny hop. Substeps
            // also preserve ceiling/ground contacts if a frame spans the apex.
            float remaining = Time.deltaTime;
            while (remaining > 0f)
            {
                float dt = Mathf.Min(remaining, 1f / 60f);
                float displacement = verticalVelocity * dt + 0.5f * gravity * dt * dt;
                verticalVelocity += gravity * dt;
                CollisionFlags flags = characterController.Move(Vector3.up * displacement);
                frameCollisionFlags |= flags;
                if ((flags & CollisionFlags.Above) != 0 && verticalVelocity > 0f)
                    verticalVelocity = 0f;
                if ((flags & CollisionFlags.Below) != 0 && verticalVelocity <= 0f)
                {
                    verticalVelocity = -2f;
                    jumpConsumed = false;
                }
                remaining -= dt;
            }
        }

        private void UpdateGroundState()
        {
            isOnGroundSurface = IsGroundSurfaceBelowPlayer();

            if (isOnGroundSurface && !wasOnGroundSurface)
            {
                jumpConsumed = false;
            }

            wasOnGroundSurface = isOnGroundSurface;
        }

        private bool IsGroundSurfaceBelowPlayer()
        {
            Vector3 origin = transform.position + Vector3.up * groundCheckOriginHeight;
            return Physics.Raycast(
                origin,
                Vector3.down,
                groundCheckDistance,
                groundLayers,
                QueryTriggerInteraction.Ignore);
        }

        private void UpdateLook()
        {
            if (cameraPivot == null || !CursorIsLocked)
            {
                return;
            }

            Vector2 lookInput = lookAction.ReadValue<Vector2>() * mouseSensitivity;
            pitch = Mathf.Clamp(pitch - lookInput.y, -maxPitch, maxPitch);
            cameraPivot.localRotation = Quaternion.Euler(pitch, 0f, 0f);
            transform.Rotate(Vector3.up * lookInput.x);
        }

        private void UpdateCrouch()
        {
            bool isBelowStandingHeight = characterController.height < standingHeight - 0.01f;
            bool wantsToCrouch = crouchAction.IsPressed() ||
                                 (isBelowStandingHeight && !HasStandingClearance());
            float minimumHeight = characterController.radius * 2f;
            float targetHeight = wantsToCrouch
                ? Mathf.Clamp(crouchHeight, minimumHeight, standingHeight)
                : standingHeight;
            float heightDifference = standingHeight - targetHeight;
            float maxDelta = crouchTransitionSpeed * Time.deltaTime;

            characterController.height = Mathf.MoveTowards(
                characterController.height,
                targetHeight,
                maxDelta);
            // Derive the center from the actual interpolated height so the capsule feet
            // stay fixed throughout crouching; independent center interpolation caused
            // floor depenetration and a false landing animation during the transition.
            characterController.center = standingCenter -
                Vector3.up * ((standingHeight - characterController.height) * 0.5f);

            if (cameraPivot != null)
            {
                cameraPivot.localPosition = Vector3.MoveTowards(
                    cameraPivot.localPosition,
                    standingCameraLocalPosition - Vector3.up * heightDifference,
                    maxDelta);
            }
        }

        private bool HasStandingClearance()
        {
            Vector3 scale = transform.lossyScale;
            float radiusScale = Mathf.Max(Mathf.Abs(scale.x), Mathf.Abs(scale.z));
            float heightScale = Mathf.Abs(scale.y);
            float radius = Mathf.Max(0.01f, (characterController.radius - characterController.skinWidth) * radiusScale);
            float height = Mathf.Max(standingHeight * heightScale, radius * 2f);
            float halfSegment = Mathf.Max(0f, height * 0.5f - radius);
            Vector3 center = transform.TransformPoint(standingCenter);
            Vector3 lowerPoint = center - transform.up * halfSegment + transform.up * characterController.skinWidth;
            Vector3 upperPoint = center + transform.up * halfSegment - transform.up * characterController.skinWidth;

            Collider[] overlaps = Physics.OverlapCapsule(
                lowerPoint,
                upperPoint,
                radius,
                ceilingLayers,
                QueryTriggerInteraction.Ignore);

            foreach (Collider overlap in overlaps)
            {
                if (overlap != characterController && !overlap.transform.IsChildOf(transform))
                {
                    return false;
                }
            }

            return true;
        }

        private void UpdateMovement()
        {
            Vector2 moveInput = moveAction.ReadValue<Vector2>();
            Vector3 movement = transform.right * moveInput.x + transform.forward * moveInput.y;
            float speed = sprintAction.IsPressed() ? walkSpeed * sprintMultiplier : walkSpeed;
            // Keep each collision sweep short enough to traverse ramps and steps
            // consistently at low render frame rates, including while sprinting.
            Vector3 velocity = movement.normalized * speed;
            float remaining = Time.deltaTime;
            while (remaining > 0f)
            {
                float dt = Mathf.Min(remaining, 1f / 60f);
                frameCollisionFlags |= characterController.Move(velocity * dt);
                remaining -= dt;
            }
        }
    }
}
