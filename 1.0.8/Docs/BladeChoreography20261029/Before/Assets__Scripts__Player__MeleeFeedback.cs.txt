using ArcaneArchive.Core;
using UnityEngine;

namespace ArcaneArchive.Player
{
    // A small additive camera layer, restored before the controller writes its pose.
    // The attack never changes aim state, global timeScale or saved input settings.
    [DefaultExecutionOrder(-200), DisallowMultipleComponent]
    public sealed class MeleeFeedback : MonoBehaviour
    {
        FirstPersonController player;
        PlayerMelee melee;
        BossEncounter encounter;
        Camera eye;
        AudioSource source;
        AudioClip lightWhoosh,heavyWhoosh,thrustWhoosh,shieldWhoosh,contact,parry;
        Vector3 baselinePosition,appliedPosition;
        Quaternion baselineRotation,appliedRotation;
        float baselineFov,appliedFov,impulse,impulseAge;
        bool applied,audioPaused;
        void Awake()
        {
            player=GetComponent<FirstPersonController>();melee=GetComponent<PlayerMelee>();encounter=GetComponent<BossEncounter>();
            eye=GetComponentInChildren<Camera>();source=gameObject.AddComponent<AudioSource>();
            source.playOnAwake=false;source.spatialBlend=0;source.volume=.24f;
            lightWhoosh=MakeSound("Sword air",.20f,0);heavyWhoosh=MakeSound("Heavy sword air",.29f,1);
            thrustWhoosh=MakeSound("Sword thrust air",.16f,2);shieldWhoosh=MakeSound("Shield shove",.18f,3);
            contact=MakeSound("Close steel contact",.16f,4);parry=MakeSound("Parry resonance",.32f,5);
        }
        public void Swing(int action,int combo)
        {
            if(!source)return;source.pitch=action==0?1.02f+combo*.055f:1;
            source.PlayOneShot(action==2?shieldWhoosh:action==3?thrustWhoosh:action==1 || action==4?heavyWhoosh:lightWhoosh,.70f);
        }
        public void Contact(bool heavy,bool perfect)
        {
            impulse=perfect?1.1f:heavy?1:.62f;impulseAge=0;
            if(source){source.pitch=1;source.PlayOneShot(perfect?parry:contact,perfect?.82f:.55f);}
        }
        public void Hurt(){impulse=.70f;impulseAge=0;}
        public void Clear()
        {
            RestoreCamera();impulse=impulseAge=0;if(source)source.Stop();audioPaused=false;
        }
        void Update()
        {
            RestoreCamera();
            bool pause=player && player.SimulationPaused;
            if(source && pause!=audioPaused){if(pause)source.Pause();else source.UnPause();audioPaused=pause;}
        }
        void LateUpdate()
        {
            if(!eye || !player || player.SimulationPaused || !melee || (encounter && encounter.IsFinished))return;
            float dt=Mathf.Min(Time.deltaTime,.05f);impulseAge+=dt;
            float decay=Mathf.Exp(-impulseAge*20)*impulse;
            if(decay<.001f)impulse=0;
            float pulse=0,roll=0,fov=0;
            if(melee.Busy)
            {
                float t=melee.ActionAge,impact=melee.ImpactTime;
                pulse=Mathf.Sin(Mathf.Clamp01(t/melee.Duration)*Mathf.PI);
                float direction=melee.Action==0 && melee.Combo==1?-1:1;
                roll=direction*Mathf.Sin(Mathf.Clamp01(t/(impact+.18f))*Mathf.PI)*(melee.Action==4?1.25f:.65f);
                if(melee.Action==3)fov=3.2f*Mathf.Sin(Mathf.Clamp01(t/melee.Duration)*Mathf.PI);
            }
            Vector3 angles=new Vector3(-decay*1.2f+Mathf.Sin(impulseAge*52)*decay*.38f,0,roll+decay*.65f);
            Vector3 shift=new Vector3(0,-pulse*.003f,-decay*.016f);
            baselinePosition=eye.transform.localPosition;baselineRotation=eye.transform.localRotation;baselineFov=eye.fieldOfView;
            appliedPosition=baselinePosition+shift;appliedRotation=baselineRotation*Quaternion.Euler(angles);appliedFov=baselineFov+fov;
            eye.transform.SetLocalPositionAndRotation(appliedPosition,appliedRotation);eye.fieldOfView=appliedFov;applied=true;
        }
        void RestoreCamera()
        {
            if(!applied || !eye)return;
            // A teleport or another owner may have replaced the camera pose already.
            if((eye.transform.localPosition-appliedPosition).sqrMagnitude<.000001f)eye.transform.localPosition=baselinePosition;
            if(Quaternion.Angle(eye.transform.localRotation,appliedRotation)<.01f)eye.transform.localRotation=baselineRotation;
            if(Mathf.Abs(eye.fieldOfView-appliedFov)<.001f)eye.fieldOfView=baselineFov;
            applied=false;
        }
        static AudioClip MakeSound(string name,float seconds,int kind)
        {
            const int rate=22050;int count=Mathf.CeilToInt(seconds*rate);var samples=new float[count];
            var random=new System.Random(171+kind);float low=0,previous=0,phase=0;
            for(int i=0;i<count;i++)
            {
                float t=i/(float)rate,u=t/seconds,noise=(float)random.NextDouble()*2-1;
                low=Mathf.Lerp(low,noise,kind==1?.09f:.24f);float high=noise-previous;previous=noise;
                float envelope=Mathf.Pow(Mathf.Sin(u*Mathf.PI),kind<4?1.6f:.65f)*Mathf.Exp(-u*(kind<4?1.4f:5));
                float frequency=kind==5?1320:kind==4?680:Mathf.Lerp(kind==1?180:360,85,u);
                phase+=frequency*2*Mathf.PI/rate;
                float tone=Mathf.Sin(phase)+.27f*Mathf.Sin(phase*2.73f);
                float value=kind>=4?(high*.13f+tone*.44f):(low*.83f+tone*.06f);
                samples[i]=Mathf.Clamp(value*envelope,-.8f,.8f);
            }
            var clip=AudioClip.Create(name,count,1,rate,false);clip.SetData(samples,0);return clip;
        }
        void OnDisable(){RestoreCamera();if(source)source.Stop();}
        void OnDestroy()
        {
            RestoreCamera();foreach(var clip in new[]{lightWhoosh,heavyWhoosh,thrustWhoosh,shieldWhoosh,contact,parry})if(clip)Destroy(clip);
            if(source)Destroy(source);
        }
    }
}
