using ArcaneArchive.Core;
using ArcaneArchive.Visuals;
using UnityEngine;

namespace ArcaneArchive.Player
{
    [DisallowMultipleComponent, RequireComponent(typeof(PlayerCombat))]
    public sealed class PlayerMelee : MonoBehaviour
    {
        public MeleeLoadout loadout;
        public SwordShieldView view;
        FirstPersonController player;
        PlayerCombat combat;
        BossEncounter encounter;
        CharacterController capsule;
        float age, regenDelay, chainGrace, inputBuffer, practiceClock;
        readonly float[] cooldowns = new float[4];
        Vector3 committedDirection;
        bool hit, queuedLight, waitForRelease = true;
        int combo;
        public float Stamina { get; private set; }
        // -1 is idle, 0 is the light chain, 1..4 are the four skills.
        public int Action { get; private set; } = -1;
        public int Combo => combo;
        public bool Busy => Action >= 0;
        public float ActionAge => age;
        public float Duration => Action > 0 ? loadout.skills[Action-1].duration : loadout.lightDuration + (combo==2?.13f:0);
        public float Progress => Busy ? Mathf.Clamp01(age/Duration) : 0;
        public bool CanGuard => !Busy || Progress > .80f;
        public bool PracticeGuarding { get; private set; }
        public uint SwingSequence { get; private set; }
        public uint HitSequence { get; private set; }
        public float Cooldown(int index) => index>=0 && index<4 ? cooldowns[index] : 0;
        public float Clock => encounter && encounter.IsActive ? encounter.BattleTime : practiceClock;
        void Awake()
        {
            player=GetComponent<FirstPersonController>(); combat=GetComponent<PlayerCombat>();
            encounter=GetComponent<BossEncounter>(); capsule=GetComponent<CharacterController>();
            ResetCombat();
        }
        public void ResetCombat()
        {
            Stamina=loadout?loadout.maxStamina:100; Action=-1; age=regenDelay=chainGrace=inputBuffer=0;
            queuedLight=hit=false; combo=0; waitForRelease=true; PracticeGuarding=false;
            System.Array.Clear(cooldowns,0,cooldowns.Length);
            if(view)view.ResetPose();
        }
        void Update()
        {
            if(!loadout || (encounter && encounter.IsActive) || player.SimulationPaused)return;
            float dt=Mathf.Min(Time.deltaTime,.1f);practiceClock+=dt;Advance(dt);
            var b=player.Bindings;
            PracticeGuarding=b!=null && player.GameplayInputEnabled && player.CursorIsLocked && !b.SuppressInputThisFrame && b.AbsorbAction.IsPressed() && CanGuard;
            ReadInput();
        }
        public void SuppressUntilReleased() { waitForRelease=true;queuedLight=false;inputBuffer=0;PracticeGuarding=false; }
        public void Advance(float dt)
        {
            if(!loadout)return;
            for(int i=0;i<4;i++)cooldowns[i]=Mathf.Max(0,cooldowns[i]-dt);
            regenDelay=Mathf.Max(0,regenDelay-dt); chainGrace=Mathf.Max(0,chainGrace-dt);
            if(!Busy && regenDelay<=0 && !combat.Guarding && !PracticeGuarding)Stamina=Mathf.Min(loadout.maxStamina,Stamina+loadout.staminaRegen*dt);
            if(!Busy)return;
            float previous=age;age+=dt;
            // A bounded CharacterController move cannot teleport through walls or the boss.
            if(Action==3)
            {
                float before=Mathf.SmoothStep(0,1,Mathf.Clamp01((previous-.10f)/.28f));
                float after=Mathf.SmoothStep(0,1,Mathf.Clamp01((age-.10f)/.28f));
                capsule.Move(committedDirection*((after-before)*2.35f));
            }
            float impact=Action>0?loadout.skills[Action-1].impactTime:(combo==2?.31f:.24f);
            if(!hit && age>=impact){hit=true;ResolveHit();}
            if(age>=Duration)
            {
                bool next=queuedLight;Action=-1;age=0;queuedLight=false;chainGrace=.55f;
                if(next)TryLightAttack();
            }
        }
        public void ReadInput()
        {
            if(!loadout || !player.Bindings)return;
            var b=player.Bindings;
            if(!player.GameplayInputEnabled || !player.CursorIsLocked || b.SuppressInputThisFrame){SuppressUntilReleased();return;}
            if(waitForRelease)
            {
                bool held=b.AttackAction.IsPressed() || b.AbsorbAction.IsPressed();
                foreach(var skill in b.MeleeSkillActions)held|=skill.IsPressed();
                if(!held)waitForRelease=false;
                return;
            }
            for(int i=0;i<4;i++)if(b.MeleeSkillActions[i].WasPressedThisFrame()){TrySkill(i);return;}
            if(b.AttackAction.WasPressedThisFrame())inputBuffer=.18f;
            inputBuffer=Mathf.Max(0,inputBuffer-Time.deltaTime);
            if(inputBuffer>0 || b.AttackAction.IsPressed())
            {
                if(Busy && Action==0 && Progress>.66f)queuedLight=true;
                else if(!Busy && TryLightAttack())inputBuffer=0;
            }
        }
        bool CanAct => loadout && !player.SimulationPaused && player.GameplayInputEnabled &&
            (!encounter || !encounter.IsActive || encounter.CanFight) && !combat.Guarding && !PracticeGuarding;
        public bool TryLightAttack()
        {
            if(!CanAct || Busy || Stamina<loadout.lightCost)return false;
            combo=chainGrace>0?(combo+1)%3:0;
            StartAction(0,loadout.lightCost);return true;
        }
        public bool TrySkill(int index)
        {
            if(index<0 || index>=4 || !CanAct || Busy)return false;
            var s=loadout.skills[index];
            if(cooldowns[index]>0 || Stamina<s.stamina)
            {if(encounter && encounter.IsActive)encounter.Feedback(cooldowns[index]>0?s.title+" · 尚在冷却":"体力不足 · 暂缓进攻",.65f);return false;}
            cooldowns[index]=s.cooldown; StartAction(index+1,s.stamina);return true;
        }
        void StartAction(int action,float cost)
        {
            Action=action;age=0;hit=queuedLight=false;Stamina-=cost;regenDelay=loadout.staminaRegenDelay;SwingSequence++;
            committedDirection=Vector3.ProjectOnPlane(player.GetComponentInChildren<Camera>().transform.forward,Vector3.up).normalized;
            if(committedDirection.sqrMagnitude<.01f)committedDirection=transform.forward;
            if(view)view.BeginSwing();
        }
        void ResolveHit()
        {
            if(!encounter || !encounter.CanFight)return;
            float damage=loadout.lightDamage*(combo==2?1.45f:1),reach=loadout.lightReach,arc=105,poise=combo==2?28:16;
            if(Action>0){var s=loadout.skills[Action-1];damage=s.damage;reach=s.reach;arc=s.arc;poise=s.poiseDamage;}
            if(Action==4)MeleeEffects.Sweep(encounter.Arena.effectRoot,transform.position+Vector3.up*.7f,committedDirection,reach,360,loadout.violetEffect,encounter);
            if(Action==1)MeleeEffects.Sweep(encounter.Arena.effectRoot,transform.position+Vector3.up*.85f,committedDirection,reach*.85f,arc,loadout.goldEffect,encounter);
            bool empowered=combat.Power>=100;
            if(empowered)damage*=1.65f;
            if(encounter.TryMeleeHit(reach,arc,damage,poise,Action==2))
            {
                HitSequence++;if(empowered)combat.ConsumePower();
                if(Action==2)combat.RestoreGuard(16);
                if(view)view.Impact(false);
            }
        }
        public void Parried() { Stamina=Mathf.Min(loadout.maxStamina,Stamina+loadout.parryRefund);if(view)view.Impact(true); }
        public void Interrupt()
        {
            Action=-1;queuedLight=false;age=0;chainGrace=0;
            if(view)view.Interrupt();
        }
    }
}
