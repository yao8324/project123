var e=UnityEngine.Object.FindFirstObjectByType<ArcaneArchive.Core.BossEncounter>();
var p=e.GetComponent<ArcaneArchive.Player.FirstPersonController>();var d=p.Dash;var m=e.GetComponent<ArcaneArchive.Player.PlayerMelee>();var c=e.Combat;var controller=e.GetComponent<CharacterController>();var menu=e.GetComponent<ArcaneArchive.UI.RecoveryMenu>();
var flags=System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance;
System.Action<string,object> set=(n,v)=>e.GetType().GetField(n,flags).SetValue(e,v);
System.Action<string,object[]> call=(n,a)=>e.GetType().GetMethod(n,flags).Invoke(e,a);
System.Action<ArcaneArchive.Core.BossEncounter.State> state=s=>call("SetState",new object[]{s});
System.Action<Vector3> place=position=>{controller.enabled=false;e.transform.SetPositionAndRotation(position,Quaternion.identity);controller.enabled=true;Physics.SyncTransforms();p.GetType().GetField("<IsGrounded>k__BackingField",flags).SetValue(p,true);};
System.Action reset=()=>{d.Cancel(true);menu.Resume();set("focusLost",false);p.SetGameplayInputEnabled(true);c.ResetCombat(e);call("ClearAttacks",new object[0]);state(ArcaneArchive.Core.BossEncounter.State.Observe);set("combatTime",0f);e.Arena.boss.position=new Vector3(160,0,0);e.Arena.boss.rotation=Quaternion.Euler(0,180,0);place(new Vector3(160,.08f,-4.5f));};
System.Action<float> step=dt=>{d.Tick(dt);CollisionFlags collision;d.Move(out collision);Physics.SyncTransforms();};
System.Action strike=()=>{set("attack",5);state(ArcaneArchive.Core.BossEncounter.State.Attack);set("stateTime",e.definition.meleeImpact);set("meleeResolved",false);call("TickMeleeStrike",new object[0]);d.ConfirmPerfect();};

var lines=new System.Collections.Generic.List<string>();
System.Action<bool,string> check=(ok,label)=>lines.Add((ok?"PASS ":"FAIL ")+label);

System.Action<float> preparePage=wallOffset=>{
    reset();
    var shotType=e.GetType().GetNestedType("Shot",System.Reflection.BindingFlags.NonPublic);
    var shots=(System.Collections.IList)e.GetType().GetField("shots",flags).GetValue(e);
    var shot=System.Activator.CreateInstance(shotType,true);
    var page=new GameObject("StaminaDodgeTest_Page");page.transform.SetParent(e.Arena.effectRoot);
    page.transform.position=e.transform.position+new Vector3(0,1,1.3f);
    shotType.GetField("go").SetValue(shot,page);
    shotType.GetField("velocity").SetValue(shot,Vector3.back*22);
    shotType.GetField("enemy").SetValue(shot,true);
    shotType.GetField("damage").SetValue(shot,10f);
    shotType.GetField("life").SetValue(shot,2f);shots.Add(shot);
    if(wallOffset!=0){
        var barrier=GameObject.CreatePrimitive(PrimitiveType.Cube);barrier.name="StaminaDodgeTest_Wall";
        barrier.transform.SetParent(e.Arena.effectRoot);
        barrier.transform.position=e.transform.position+new Vector3(0,1,wallOffset);
        barrier.transform.localScale=new Vector3(1,3,.1f);
    }
    Physics.SyncTransforms();d.TryStart(Vector2.left);step(.08f);
};

uint sequence=d.PerfectSequence;call("TickShots",new object[]{.1f});d.ConfirmPerfect();
var remainingPages=(System.Collections.IList)e.GetType().GetField("shots",flags).GetValue(e);
check(d.PerfectSequence==sequence+1 && remainingPages.Count==0 && c.Health==e.definition.playerHealth,"page crosses old capsule then hits wall in same step: valid dodge survives projectile consumption");
var wallToRemove=GameObject.Find("StaminaDodgeTest_Wall");if(wallToRemove)UnityEngine.Object.DestroyImmediate(wallToRemove);
preparePage(.5f);

var report=string.Join("\n",lines.ToArray());
var folder=System.IO.Path.Combine(System.IO.Directory.GetParent(Application.dataPath).FullName,"Docs/StaminaDodge20261030");
System.IO.File.AppendAllText(folder+"/MinimalChecks.txt",report+"\n",new System.Text.UTF8Encoding(false));return report;
